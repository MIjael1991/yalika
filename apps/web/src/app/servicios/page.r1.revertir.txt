const STATUS: Record<
  (typeof SERVICES)[number]['status'],
  string
> = {
  existing: 'Módulo existente',
  review: 'Requiere validación funcional',
  development: 'En desarrollo',
  planned: 'Próximamente',
};

export default async function ServicesPage() {
  const apiUrl = (
    process.env.ORION_API_URL ?? 'http://localhost:3000'
  ).replace(/\/+$/, '');

  const cookieHeader = (await cookies()).toString();

  let response: Response;

  try {
    response = await fetch(`${apiUrl}/auth/me`, {
      headers: { cookie: cookieHeader },
      cache: 'no-store',
    });
  } catch {
    redirect('/login');
  }

  if (!response.ok) redirect('/login');

  let session: Session;

  try {
    session = (await response.json()) as Session;
  } catch {
    redirect('/login');
  }

  const modules = session.user?.modules ?? [];

  if (!modules.includes('all')) {
    redirect(
      modules.includes('surveys')
        ? '/encuestas'
        : '/login'
    );
  }

  return (
    <main className="min-h-screen bg-slate-950 px-4 py-10 text-slate-100 sm:px-8">
      <div className="mx-auto max-w-6xl">

        <p className="mb-2 text-sm font-semibold uppercase tracking-[0.3em] text-cyan-400">
          Yalika
        </p>

        <h1 className="text-3xl font-bold sm:text-4xl">
          Mis servicios
        </h1>

        <p className="mt-3 max-w-2xl text-slate-300">
          Hola, {session.user?.name?.trim().split(/\s+/)[0] || 'bienvenido'}.
          Aquí puedes conocer los servicios existentes y los que estamos preparando.
        </p>

        <div className="mt-8 rounded-2xl border border-cyan-700/40 bg-cyan-950/30 p-5 text-sm text-cyan-100">
          Esta es la primera versión del catálogo.
          Consultar un servicio no lo activa, no concede permisos
          ni crea cobros.
        </div>

        <div className="mt-8 grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
          {SERVICES.map((service) => (
            <article
              key={service.id}
              className="flex min-h-52 flex-col rounded-2xl border border-slate-700 bg-slate-900 p-5"
            >
              <span className="text-xs font-semibold uppercase tracking-wide text-cyan-300">
                {STATUS[service.status]}
              </span>

              <h2 className="mt-3 text-xl font-bold">
                {service.name}
              </h2>

              <p className="mt-2 flex-1 text-sm text-slate-300">
                {service.summary}
              </p>

              {service.status === 'existing' && service.href ? (
                <Link
                  href={service.href}
                  className="mt-5 inline-flex min-h-11 items-center justify-center rounded-xl bg-cyan-500 px-4 text-sm font-bold text-slate-950 hover:bg-cyan-400"
                >
                  Abrir servicio
                </Link>
              ) : (
                <p className="mt-5 rounded-xl border border-slate-700 px-4 py-3 text-center text-xs text-slate-400">
                  {service.status === 'review'
                    ? 'Disponible tras verificar su funcionamiento'
                    : 'Todavía no habilitado'}
                </p>
              )}
            </article>
          ))}
        </div>

        <p className="mt-8 text-xs text-slate-400">
          Encuestas conserva su publicación y dirección actuales.
          Este catálogo no modifica su servidor.
        </p>

      </div>
    </main>
  );
}
